Sub Выделение_акций()
    
    Dim i As Long
    Dim lastRow As Long
    Dim valB As String
    Dim valB_Original As String
    Dim valC As Variant
    Dim startRow As Long
    
    ' ===== Цвета =====
    Dim COLOR_GREEN As Long
    Dim COLOR_YELLOW As Long
    
    COLOR_GREEN = RGB(144, 238, 144)   ' Светло-зелёный
    COLOR_YELLOW = RGB(255, 255, 153)  ' Светло-жёлтый
    
    ' Отключаем обновление экрана
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    ' Находим последнюю строку в столбце B
    lastRow = Cells(Rows.Count, 2).End(xlUp).Row
    
    ' Определяем строку, с которой начинаются данные
    startRow = 1
    For i = 1 To lastRow
        valB = Trim(Cells(i, 2).Value)
        If valB <> "" And LCase(valB) <> "артикул" And Not IsNumeric(valB) Then
            startRow = i
            Exit For
        End If
    Next i
    
    ' Проходим по всем строкам СНИЗУ ВВЕРХ
    For i = lastRow To startRow Step -1
        valB = LCase(Cells(i, 2).Value)
        valB_Original = Cells(i, 2).Value
        valC = Cells(i, 3).Value
        
        ' ============================================================
        ' УСЛОВИЕ 1: Удаляем ВСЕ строки с "Мит Гарден"
        ' ============================================================
        If InStr(valB, "мит гарден") > 0 Then
            Rows(i).Delete
            GoTo NextRow
        End If
        
        ' ============================================================
        ' УСЛОВИЕ 2: Проверяем исключения в столбце C
        ' ============================================================
        If IsEmpty(valC) Or valC = "" Then
            GoTo NextRow
        End If
        
        If CStr(valC) = "21261016" Or CStr(valC) = "00175" Or CStr(valC) = "175" Then
            GoTo NextRow
        End If
        
        ' ============================================================
        ' ИСКЛЮЧЕНИЯ (не выделять)
        ' ============================================================
        If InStr(valB, "бест диннер монопротеин") > 0 Or InStr(valB, "бест диннер") > 0 Then
            GoTo NextRow
        End If
        
        If InStr(valB_Original, "УПАК") > 0 Then
            GoTo NextRow
        End If
        
        ' ============================================================
        ' УСЛОВИЕ 3: НОВЫЕ БРЕНДЫ (зелёный)
        ' ============================================================
        If InStr(valB, "монопротеин стерил") > 0 Then
            Cells(i, 2).Interior.Color = COLOR_GREEN
            GoTo NextRow
        End If
        
        If InStr(valB, "оптином") > 0 Then
            Cells(i, 2).Interior.Color = COLOR_GREEN
            GoTo NextRow
        End If
        
        If InStr(valB, "прошейп") > 0 Or InStr(valB, "про шейп") > 0 Then
            Cells(i, 2).Interior.Color = COLOR_GREEN
            GoTo NextRow
        End If
        
        ' ============================================================
        ' УСЛОВИЕ 4: АКЦИИ (жёлтый)
        ' ============================================================
        If InStr(valB, "акция") > 0 Or InStr(LCase(Cells(i, 3).Value), "акция") > 0 Then
            Cells(i, 2).Interior.Color = COLOR_YELLOW
            GoTo NextRow
        End If
        
        If InStr(valB_Original, "АКЦ") > 0 Then
            Cells(i, 2).Interior.Color = COLOR_YELLOW
            GoTo NextRow
        End If
        
        If InStr(valB_Original, "АК ") > 0 And InStr(valB_Original, "АКЦ") = 0 Then
            Cells(i, 2).Interior.Color = COLOR_YELLOW
            GoTo NextRow
        End If
        
NextRow:
    Next i
    
    ' ============================================================
    ' СОХРАНЕНИЕ ФАЙЛА
    ' ============================================================
    On Error Resume Next
    ActiveWorkbook.Save
    On Error GoTo 0
    
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    
    ' ============================================================
    ' УВЕДОМЛЕНИЕ И ЗАКРЫТИЕ
    ' ============================================================
    MsgBox "Готово!", vbInformation, "Выполнено"
    
    ' Закрываем Excel
    Application.Quit

End Sub
