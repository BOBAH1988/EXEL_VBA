Sub Очистка_Прайса()

    ' =====================================================
    ' ОПТИМИЗАЦИЯ ПРОИЗВОДИТЕЛЬНОСТИ
    ' =====================================================
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    
    ' =====================================================
    ' ПРОВЕРКА НА ПУСТОЙ ЛИСТ
    ' =====================================================
    If Application.WorksheetFunction.CountA(ActiveSheet.Cells) = 0 Then
        MsgBox "Лист пуст! Операция отменена.", vbExclamation, "Ошибка"
        GoTo ExitSub
    End If
    
    ' =====================================================
    ' ОБРАБОТКА ОШИБОК
    ' =====================================================
    On Error GoTo ErrorHandler
    
    ' =====================================================
    ' ШАГ 1: Удаляем строки с 1 по 10
    ' =====================================================
    If ActiveSheet.Rows.Count >= 10 Then
        Rows("1:10").Delete Shift:=xlUp
    Else
        MsgBox "На листе меньше 10 строк. Проверьте данные.", vbExclamation, "Предупреждение"
        GoTo ExitSub
    End If
    
    ' =====================================================
    ' ШАГ 2: Проверяем столбец G на наличие формул (ВСЕ СТРОКИ)
    ' =====================================================
    If ActiveSheet.Columns.Count >= 7 Then
        Dim hasFormula As Boolean
        hasFormula = False
        Dim checkRow As Long
        Dim lastRowG As Long
        
        ' Находим последнюю заполненную строку в столбце G
        On Error Resume Next
        lastRowG = Cells(Rows.Count, 7).End(xlUp).Row
        On Error GoTo ErrorHandler
        
        ' Если данных нет, проверяем хотя бы 1500 строк
        If lastRowG < 1 Then lastRowG = 1500
        
        ' Проверяем все строки столбца G
        For checkRow = 1 To lastRowG
            If Cells(checkRow, 7).HasFormula Then
                hasFormula = True
                Exit For
            End If
        Next checkRow
        
        ' Если есть формулы - удаляем столбец G
        If hasFormula Then
            Columns("G:G").Delete Shift:=xlToLeft
        End If
    End If

    ' =====================================================
    ' ШАГ 3: Удаляем столбцы D и E
    ' =====================================================
    If ActiveSheet.Columns.Count >= 5 Then
        Columns("D:E").Delete Shift:=xlToLeft
    End If

    ' =====================================================
    ' ШАГ 4: Сокращаем ширину столбца B на 25%
    ' =====================================================
    Dim currentWidth As Double
    currentWidth = Columns("B").ColumnWidth
    If currentWidth > 0 Then
        Columns("B").ColumnWidth = currentWidth * 0.75
    End If

    ' =====================================================
    ' ШАГ 5: Отключаем перенос текста в столбце B
    ' =====================================================
    If ActiveSheet.Columns.Count >= 2 Then
        Columns("B").WrapText = False
    End If

    ' =====================================================
    ' ШАГ 6: Удаляем строки с пустыми значениями в столбце D
    ' =====================================================
    Dim lastRow As Long
    Dim i As Long
    Dim deleteRange As Range
    
    If ActiveSheet.Columns.Count >= 4 Then
    
        lastRow = Cells(Rows.Count, 3).End(xlUp).Row
        
        If lastRow < 2 Then lastRow = 2
        
        For i = 2 To lastRow
            If IsEmpty(Cells(i, 4)) Or Cells(i, 4).Value = "" Then
                If deleteRange Is Nothing Then
                    Set deleteRange = Rows(i)
                Else
                    Set deleteRange = Union(deleteRange, Rows(i))
                End If
            End If
        Next i
        
        If Not deleteRange Is Nothing Then
            deleteRange.Delete
        End If
        
    End If

    ' =====================================================
    ' ШАГ 7: Выравниваем столбец D по ширине значений
    ' =====================================================
    If ActiveSheet.Columns.Count >= 4 Then
        On Error Resume Next
        Columns("D").AutoFit
        On Error GoTo ErrorHandler
    End If

    ' =====================================================
    ' ШАГ 8: Увеличиваем масштаб листа до 150%
    ' =====================================================
    On Error Resume Next
    ActiveWindow.Zoom = 150
    On Error GoTo ErrorHandler

    ' =====================================================
    ' ШАГ 9: Выравниваем столбец C по правому краю
    ' =====================================================
    If ActiveSheet.Columns.Count >= 3 Then
        Columns("C").HorizontalAlignment = xlRight
    End If

    ' =====================================================
    ' ШАГ 10: Возвращаем курсор на ячейку B2 (БЕЗ СМЕЩЕНИЯ ОБЗОРА)
    ' =====================================================
    On Error Resume Next
    
    ' Активируем лист
    ActiveSheet.Activate
    
    ' Выбираем ячейку B2 (курсор перемещается)
    Range("B2").Select
    
    ' НО прокрутку оставляем в начале листа (A1)
    ' Прокручиваем окно так, чтобы A1 был виден
    ActiveWindow.ScrollRow = 1
    ActiveWindow.ScrollColumn = 1
    
    On Error GoTo ErrorHandler

    ' =====================================================
    ' ЗАВЕРШЕНИЕ: Сообщение об успехе
    ' =====================================================
    Application.StatusBar = "Обработка завершена успешно!"
    
    GoTo ExitSub

    ' =====================================================
    ' ОБРАБОТЧИК ОШИБОК
    ' =====================================================
ErrorHandler:
    MsgBox "Произошла ошибка: " & Err.Description & vbCrLf & _
           "Код ошибки: " & Err.Number, vbCritical, "Ошибка выполнения"
    
    If Not deleteRange Is Nothing Then Set deleteRange = Nothing

    ' =====================================================
    ' ВОССТАНОВЛЕНИЕ НАСТРОЕК
    ' =====================================================
ExitSub:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.DisplayAlerts = True

End Sub
