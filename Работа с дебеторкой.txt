Sub Обработка_Дебиторки()
    ' ============================================================
    ' Назначение: Автоматическая очистка и подготовка файла дебиторской задолженности
    ' Действия: удаление служебных строк/столбцов, удаление закрытых клиентов,
    '           удаление строк по сумме долга, выделение строк, настройка ширины и масштаба
    ' Горячая клавиша: Ctrl+Shift+J (Win) / Cmd+Shift+J (Mac)
    ' ============================================================
    
    ' ============================================================
    ' ОПТИМИЗАЦИЯ
    ' ============================================================
    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    Application.DisplayAlerts = False
    
    ' ============================================================
    ' ПРОВЕРКА НА ПУСТОЙ ЛИСТ
    ' ============================================================
    If Application.WorksheetFunction.CountA(ActiveSheet.Cells) = 0 Then
        MsgBox "Лист пуст! Операция отменена.", vbExclamation, "Ошибка"
        GoTo ExitSub
    End If
    
    ' ============================================================
    ' ОБРАБОТКА ОШИБОК
    ' ============================================================
    On Error GoTo ErrorHandler
    
    ' ============================================================
    ' ШАГ 1: УДАЛЕНИЕ СЛУЖЕБНЫХ СТРОК И СТОЛБЦОВ
    ' ============================================================
    
    ' Удаляем строки
    Rows("1:3").Delete Shift:=xlUp
    
    ' Удаляем столбцы (справа налево, чтобы нумерация не сбивалась)
    Columns("G:G").Delete Shift:=xlToLeft
    Columns("E:E").Delete Shift:=xlToLeft
    Columns("D:D").Delete Shift:=xlToLeft
    Columns("B:B").Delete Shift:=xlToLeft
    
    ' ============================================================
    ' ШАГ 2: УДАЛЕНИЕ ЗАКРЫТЫХ КЛИЕНТОВ ПО СПИСКУ
    ' ============================================================
    
    Dim lastRowA As Long, j As Long, deleteRangeClients As Range, valA As String
    Dim clientsToDelete As Variant
    Dim k As Long
    
    ' Список клиентов для удаления
    clientsToDelete = Array("б2693", "01063", "a1001", "a0145", "03665", "б6926", "б7092", "03286", "05752")
    
    On Error Resume Next
    lastRowA = Cells(Rows.Count, 1).End(xlUp).Row
    On Error GoTo ErrorHandler
    
    If lastRowA >= 2 Then
        For j = lastRowA To 2 Step -1
            valA = LCase(CStr(Cells(j, 1).Value))
            Dim deleteClient As Boolean: deleteClient = False
            
            ' Проверяем по массиву
            For k = LBound(clientsToDelete) To UBound(clientsToDelete)
                If InStr(valA, clientsToDelete(k)) > 0 Then
                    deleteClient = True
                    Exit For
                End If
            Next k
            
            If deleteClient Then
                If deleteRangeClients Is Nothing Then
                    Set deleteRangeClients = Rows(j)
                Else
                    Set deleteRangeClients = Union(deleteRangeClients, Rows(j))
                End If
            End If
        Next j
        
        If Not deleteRangeClients Is Nothing Then
            deleteRangeClients.Delete
            Set deleteRangeClients = Nothing
        End If
    End If
    
    ' ============================================================
    ' ШАГ 3: УДАЛЕНИЕ СТРОК ПО УСЛОВИЮ (C < -28 или C > 30)
    ' ============================================================
    
    Dim lastRow As Long, i As Long, deleteRange As Range, valC As Variant
    Dim deletedRowsCount As Long
    Dim highlightedRowsCount As Long
    
    On Error Resume Next
    lastRow = Cells(Rows.Count, 3).End(xlUp).Row
    On Error GoTo ErrorHandler
    
    If lastRow >= 2 Then
        For i = lastRow To 2 Step -1
            valC = Cells(i, 3).Value
            If IsNumeric(valC) Then
                If valC < -28 Or valC > 30 Then
                    If deleteRange Is Nothing Then
                        Set deleteRange = Rows(i)
                    Else
                        Set deleteRange = Union(deleteRange, Rows(i))
                    End If
                    deletedRowsCount = deletedRowsCount + 1
                End If
            End If
        Next i
        
        If Not deleteRange Is Nothing Then
            deleteRange.Delete
            Set deleteRange = Nothing
        End If
    End If
    
    ' ============================================================
    ' ШАГ 4: ВЫДЕЛЕНИЕ СТРОК (A, B, C) СВЕТЛО-КРАСНЫМ, ЕСЛИ C > 0
    ' ============================================================
    
    On Error Resume Next
    lastRow = Cells(Rows.Count, 3).End(xlUp).Row
    On Error GoTo ErrorHandler
    
    If lastRow >= 2 Then
        For i = 2 To lastRow
            valC = Cells(i, 3).Value
            If IsNumeric(valC) Then
                If valC > 0 Then
                    Range("A" & i & ":C" & i).Interior.Color = RGB(255, 200, 200)
                    highlightedRowsCount = highlightedRowsCount + 1
                End If
            End If
        Next i
    End If
    
    ' ============================================================
    ' ШАГ 5: НАСТРОЙКА ОТОБРАЖЕНИЯ
    ' ============================================================
    
    On Error Resume Next
    Columns("B").AutoFit
    If Not ActiveWindow Is Nothing Then
        ActiveWindow.Zoom = 150
    End If
    
    ' Показываем таблицу и ставим курсор в B2 (с обзором на A1)
    ActiveSheet.Activate
    ActiveWindow.ScrollRow = 1
    ActiveWindow.ScrollColumn = 1
    Range("B2").Select
    On Error GoTo ErrorHandler
    
    ' ============================================================
    ' ШАГ 6: ПОКАЗЫВАЕМ РЕЗУЛЬТАТ В СТАТУСНОЙ СТРОКЕ (БЕЗ БЕЛОГО ФОНА!)
    ' ============================================================
    
    ' Формируем сообщение
    Dim msg As String
    msg = "✅ Обработка дебиторки завершена! "
    msg = msg & "Удалено строк: " & deletedRowsCount & " | "
    msg = msg & "Выделено красным: " & highlightedRowsCount & " | "
    msg = msg & "Закрытых клиентов удалено: " & UBound(clientsToDelete) + 1
    
    ' Показываем в статусной строке (внизу Excel) - ТАБЛИЦА ВИДНА!
    Application.StatusBar = msg
    
    ' ИЛИ показываем в строке формул (ещё один способ)
    ' ActiveCell.Formula = msg  ' Раскомментировать, если нужно вставить в ячейку
    
    ' Включаем обновление экрана, чтобы таблица стала видна
    Application.ScreenUpdating = True
    
    ' Небольшая задержка, чтобы пользователь увидел таблицу
    ' (можно убрать, если не нужно)
    ' Dim pause As Double
    ' pause = Timer
    ' Do While Timer < pause + 0.5
    '     DoEvents
    ' Loop
    
    GoTo ExitSub

    ' ============================================================
    ' ОБРАБОТЧИК ОШИБОК
    ' ============================================================
ErrorHandler:
    ' Показываем ошибку, но без белого фона
    Application.StatusBar = "❌ Ошибка: " & Err.Description
    Application.ScreenUpdating = True
    
    ' Очищаем объекты
    If Not deleteRangeClients Is Nothing Then Set deleteRangeClients = Nothing
    If Not deleteRange Is Nothing Then Set deleteRange = Nothing

    ' ============================================================
    ' ВОССТАНОВЛЕНИЕ НАСТРОЕК
    ' ============================================================
ExitSub:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    
    ' Если хотите, чтобы сообщение исчезло через 10 секунд (необязательно):
    ' Application.OnTime Now + TimeValue("00:00:10"), "ClearStatusBar"

End Sub

' ============================================================
' ДОПОЛНИТЕЛЬНАЯ ФУНКЦИЯ ДЛЯ ОЧИСТКИ СТАТУСНОЙ СТРОКИ (опционально)
' ============================================================
Sub ClearStatusBar()
    Application.StatusBar = False
End Sub
